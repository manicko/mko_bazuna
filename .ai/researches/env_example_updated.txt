# .env.example — Environment variables template for Mko Bazuna
# Copy this file to .env (at the repository root, NOT inside src/backend/) and
# fill in values before starting containers. NEVER commit .env to git.

# Django
# Generate a secure key: python -c "from django.core.management.utils import get_random_secret_key; print(get_random_secret_key())"
DJANGO_SECRET_KEY=<generate-with-django-secret-key-generator>
DEBUG=False
ALLOWED_HOSTS=localhost,127.0.0.1,0.0.0.0

# PostgreSQL Database
# DATABASE_URL is the single source of truth for DB config (12-factor app)
# Format: postgres://user:password@host:port/database
DATABASE_URL=postgres://bazuna_user:your-password@localhost:5432/bazuna_db

# PostgreSQL discrete variables (used by pg_dump/pg_restore CLI tools)
POSTGRES_USER=bazuna_user
POSTGRES_DB=bazuna_db
POSTGRES_PASSWORD=your-password

# Telegram Bot username for contact deep-links (without @)
BOT_USERNAME=<your-bot-username>

# Telegram Bot
# Get your bot token from @BotFather
BOT_TOKEN=<your-bot-token-from-botfather>

# TLS Certificates (production nginx)
# Mount point for TLS certificates (fullchain.pem / privkey.pem)
TLS_CERT_PATH=/etc/nginx/certs

# Plausible Analytics (cookieless, no consent banner needed)
# Set to your Plausible host, e.g., "analytics.yourdomain.com" or "plausible.io"
PLAUSIBLE_HOST=

# Django Admin User (for site management)
# Note: Admin login uses telegram_id, so enter the telegram_id value as the username.

ADMIN_PASSWORD=  # Set in production via secret management
